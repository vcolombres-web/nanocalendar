rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function usuarioPrivado() {
      return request.auth != null
        && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    // Agenda compartida (la usa auth.js al iniciar sesión).
    // Solo usuarios con sesión (no anónimos).
    match /sharedAgenda/{documento} {
      allow read, write: if usuarioPrivado();
    }

    // Agenda privada de cada usuario
    match /data/users/{userId}/{document=**} {
      allow read, write: if usuarioPrivado()
        && request.auth.uid == userId;
    }

    // Los clientes pueden consultar los horarios.
    // Solo usuarios privados pueden modificarlos.
    match /publicConfig/{documento} {
      allow read: if request.auth != null;
      allow write: if usuarioPrivado();
    }

    // Cualquier visitante puede crear una reserva anónima.
    // Solo usuarios privados pueden verla/modificarla.
    match /publicReservations/{reserva} {
      allow create: if request.auth != null
        && request.auth.token.firebase.sign_in_provider == 'anonymous';

      allow read, update, delete: if usuarioPrivado();
    }

    // Bloquear cualquier otra cosa
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
