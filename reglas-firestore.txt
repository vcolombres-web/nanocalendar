rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function usuarioPrivado() {
      return request.auth != null
        && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    match /sharedAgenda/{documento} {
      allow read, write: if usuarioPrivado();
    }

    match /data/users/{userId}/{document=**} {
      allow read, write: if usuarioPrivado()
        && request.auth.uid == userId;
    }

    match /publicConfig/{documento} {
      allow read: if request.auth != null;
      allow write: if usuarioPrivado();
    }

    match /publicReservations/{reserva} {
      allow create: if request.auth != null
        && request.auth.token.firebase.sign_in_provider == 'anonymous';
      allow delete: if request.auth != null
        && request.auth.token.firebase.sign_in_provider == 'anonymous';
      allow read, update: if usuarioPrivado();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
